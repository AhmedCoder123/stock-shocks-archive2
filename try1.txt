            const [users, setUsers] = useState([]);

            const [presets, setPresets] = useState([]);
            const [newsFeed, setNewsFeed] = useState('');
            const [isTradingLocked, setIsTradingLocked] = useState(false);
            const [transactionLogs, setTransactionLogs] = useState([]);
            
            const [currentUser, setCurrentUser] = useState(null);
            const [loginError, setLoginError] = useState('');
            const [loading, setLoading] = useState(true);
            const [defaultStartingBalance, setDefaultStartingBalance] = useState(10000);
            
            const [currentSessionToken, setCurrentSessionToken] = useState(null);
            const [forceSignoutReason, setForceSignoutReason] = useState(null);

            const getPublicDoc = (collectionName, docId) => db.doc(`artifacts/${EVENT_ID}/public/data/${collectionName}/${docId}`);
            const getPublicColl = (collectionName) => db.collection(`artifacts/${EVENT_ID}/public/data/${collectionName}`);

            // 1. SILENT ANONYMOUS FIREBASE AUTH & REAL-TIME LISTENERS
            useEffect(() => {
                const initAuthAndStreams = async () => {
                    try {
                        await auth.signInAnonymously();
                        
                        const metadataDoc = await getPublicDoc('config', 'event_state').get();
                        if (!metadataDoc.exists) {
                            await getPublicDoc('config', 'event_state').set({ 
                                defaultStartingBalance: 10000, 
                                newsFeed: '',
                                isTradingLocked: false 
                            });
                            
                            for (const s of INITIAL_STOCKS) {
                                await getPublicDoc('stocks', s.symbol).set(s);
                            }
                            
                            await registerNewTeam('team1', 'Alpha Squad', '123', 10000);
                            await registerNewTeam('team2', 'Beta Bulls', '123', 10000);

                            await savePreset("Round 1 Opening", { NEXA: 100, VRTX: 150, QDYN: 50, AERO: 200, ECHO: 25 });
                            await savePreset("Market Half-time", { NEXA: 120, VRTX: 90, QDYN: 85, AERO: 180, ECHO: 15 });
                        }

                        // --- ATTACH CLOUD STREAM LISTENERS ---
                        
                        // Sync Stocks
                        getPublicColl('stocks').onSnapshot(snap => {
                            const stocksData = [];
                            snap.forEach(doc => stocksData.push(doc.data()));
                            setStocks(stocksData);
                        });

                        // Sync Presets Templates
                        getPublicColl('presets').onSnapshot(snap => {
                            const presetsData = [];
                            snap.forEach(doc => presetsData.push({ id: doc.id, ...doc.data() }));
                            setPresets(presetsData);
                        });

                        // Sync Config (News Feed / Lock Status / Settings)
                        getPublicDoc('config', 'event_state').onSnapshot(doc => {
                            if (doc.exists) {
                                const data = doc.data();
                                setNewsFeed(data.newsFeed || '');
                                setIsTradingLocked(!!data.isTradingLocked);
                                setDefaultStartingBalance(data.defaultStartingBalance || 10000);
                            }
                        });

                        // Sync Users List
                        db.collection(`artifacts/${EVENT_ID}/users`).onSnapshot(snap => {
                            const usersData = [];
                            snap.forEach(doc => { usersData.push({ id: doc.id, ...doc.data() }); });
                            setUsers(usersData);
                            setLoading(false);
                        });

                        // Sync Logs
                        getPublicColl('logs').onSnapshot(snap => {
                            const logs = [];
                            snap.forEach(doc => logs.push(doc.data()));
                            logs.sort((a,b) => b.timestamp - a.timestamp);
                            setTransactionLogs(logs);
                        });

                    } catch (e) {
                        console.error("Initialization Failed: ", e);
                        setLoading(false);
                    }
                };

                initAuthAndStreams();
            }, []);

            // 2. ACTIVE SESSION RESTRICTION AUDITOR
            useEffect(() => {
                if (!currentUser || currentUser.isAdmin) return;

                const unsubscribe = db.doc(`artifacts/${EVENT_ID}/users/${currentUser.id}`).onSnapshot(doc => {
                    if (doc.exists) {
                        const data = doc.data();
                        if (data.activeSessionToken && currentSessionToken && data.activeSessionToken !== currentSessionToken) {
                            setForceSignoutReason("Logged Out: Account accessed on another device or tab. Multi-session concurrency is restricted.");
                        }
                    }
                });

                return () => unsubscribe();
            }, [currentUser, currentSessionToken]);

            // --- GLOBAL CLOUD DATA MODIFIERS ---

            const toggleTradingLock = async () => {
                await getPublicDoc('config', 'event_state').update({
                    isTradingLocked: !isTradingLocked
                });
            };

            const registerNewTeam = async (id, name, password, balance) => {
                const userRef = db.doc(`artifacts/${EVENT_ID}/users/${id}`);
                await userRef.set({
                    id, name, password, balance, portfolio: {}, activeSessionToken: ""
                });
            };

            const pushScenario = async (scenario) => {
                const batch = db.batch();
                stocks.forEach(stock => {
                    const multiplier = scenario.impact[stock.symbol];
                    if (multiplier) {
                        const stockRef = getPublicDoc('stocks', stock.symbol);
                        batch.update(stockRef, {
                            prevPrice: stock.price,
                            price: Number((stock.price * multiplier).toFixed(2))
                        });
                    }
                });
                const eventStateRef = getPublicDoc('config', 'event_state');
                batch.update(eventStateRef, { newsFeed: scenario.news });
                await batch.commit();
            };

            const pushCustomNews = async (headline) => {
                await getPublicDoc('config', 'event_state').update({ newsFeed: headline });
            };

            const pushManualStockPrice = async (symbol, newPrice) => {
                const currentStock = stocks.find(s => s.symbol === symbol);
                await getPublicDoc('stocks', symbol).update({
                    prevPrice: currentStock ? currentStock.price : newPrice,
                    price: newPrice
                });
            };

            const pushManualTeamBalance = async (teamId, balance) => {
                await db.doc(`artifacts/${EVENT_ID}/users/${teamId}`).update({ balance });
            };

            const savePreset = async (name, prices) => {
                const presetRef = getPublicColl('presets').doc();
                await presetRef.set({ name, prices });
            };

            const applyPreset = async (preset) => {
                const batch = db.batch();
                Object.entries(preset.prices || {}).map(([symbol, targetPrice]) => {
                    const stockRef = getPublicDoc('stocks', symbol);
                    const currentStock = stocks.find(s => s.symbol === symbol);
                    batch.update(stockRef, {
                        prevPrice: currentStock ? currentStock.price : targetPrice,
                        price: targetPrice
                    });
                });
                await batch.commit();
            };

            const deletePreset = async (id) => {
                await getPublicColl('presets').doc(id).delete();
            };

            const executeTrade = async (teamId, stock, qty, type, callback) => {
                try {
                    // SERVER-SIDE GUARD RULES FOR LOCK STATE VERIFICATION
                    const configSnap = await getPublicDoc('config', 'event_state').get();
                    if (configSnap.exists && configSnap.data().isTradingLocked) {
                        throw "Order Rejected: Market execution is frozen by administration.";
                    }

                    const userRef = db.doc(`artifacts/${EVENT_ID}/users/${teamId}`);
                    await db.runTransaction(async (transaction) => {
                        const userDoc = await transaction.get(userRef);
                        if (!userDoc.exists) throw "Team profile structural record missing.";

                        const userData = userDoc.data();
                        if (userData.activeSessionToken && userData.activeSessionToken !== currentSessionToken) {
                            throw "Security Lockout: Session running context dropped.";
                        }

                        const currentBalance = userData.balance;
                        const portfolio = { ...userData.portfolio };
                        const totalCost = stock.price * qty;

                        if (type === 'BUY') {
                            if (currentBalance < totalCost) throw "Insufficient cash balance!";
                            const newBalance = currentBalance - totalCost;
                            const existingStock = portfolio[stock.symbol];
                            if (existingStock) {
                                const totalValue = (existingStock.qty * existingStock.avgPrice) + totalCost;
                                portfolio[stock.symbol] = { qty: existingStock.qty + qty, avgPrice: totalValue / (existingStock.qty + qty) };
                            } else {
                                portfolio[stock.symbol] = { qty, avgPrice: stock.price };
                            }
                            transaction.update(userRef, { balance: newBalance, portfolio });
                        } else {
                            const existingStock = portfolio[stock.symbol];
                            if (!existingStock || existingStock.qty < qty) throw "Insufficient assets volume.";
                            const newBalance = currentBalance + totalCost;
                            portfolio[stock.symbol].qty -= qty;
                            if (portfolio[stock.symbol].qty === 0) delete portfolio[stock.symbol];
                            transaction.update(userRef, { balance: newBalance, portfolio });
                        }
                    });

                    const logRef = getPublicColl('logs').doc();
                    const now = new Date();
                    await logRef.set({
                        time: now.toTimeString().split(' ')[0],
                        timestamp: Date.now(),
                        teamId,
                        teamName: users.find(u => u.id === teamId)?.name || teamId,
                        type, symbol: stock.symbol, qty, price: stock.price
                    });

                    callback(null, `Completed: ${type} ${qty} units of ${stock.symbol}!`);
                } catch (e) {
                    callback(e.toString());
                }
            };

            const handleLogin = async (id, password) => {
                setLoginError('');
                const cleanId = id.trim().toLowerCase();

                if (cleanId === ADMIN_CONFIG.username && password === ADMIN_CONFIG.password) {
                    setCurrentUser({ id: 'admin', name: 'Administrator', isAdmin: true });
                    return;
                }

                const user = users.find(u => u.id === cleanId && u.password === password);
                if (user) {
                    const sessionToken = Math.random().toString(36).substring(2, 15);
                    try {
                        await db.doc(`artifacts/${EVENT_ID}/users/${cleanId}`).update({ activeSessionToken: sessionToken });
                        setCurrentSessionToken(sessionToken);
                        setForceSignoutReason(null);
                        setCurrentUser({ ...user, isAdmin: false });
                    } catch (e) {
                        setLoginError('Cloud database initialization session timeout error.');
                    }
                } else {
                    setLoginError('Access Denied: Invalid identification token credentials.');
                }
            };

            const handleLogout = () => {
                setCurrentUser(null);
                setCurrentSessionToken(null);
                setForceSignoutReason(null);
                setLoginError('');
            };

            if (loading) {
                return (
                    <div className="min-h-screen bg-slate-900 flex flex-col items-center justify-center">
                        <CloudLoadingIcon />
                        <p className="mt-4 text-indigo-400 font-semibold tracking-wide animate-pulse">Establishing Secure Database Uplink...</p>
                    </div>
                );
            }

            if (!currentUser) return <LoginScreen onLogin={handleLogin} error={loginError} />;

            if (currentUser.isAdmin) {
                return (
                    <AdminDashboard 
                        users={users} defaultStartingBalance={defaultStartingBalance} setDefaultStartingBalance={setDefaultStartingBalance} 
                        onLogout={handleLogout} stocks={stocks} isTradingLocked={isTradingLocked} toggleTradingLock={toggleTradingLock}
                        pushScenario={pushScenario} pushCustomNews={pushCustomNews} pushManualStockPrice={pushManualStockPrice} 
                        pushManualTeamBalance={pushManualTeamBalance} transactionLogs={transactionLogs} registerNewTeam={registerNewTeam}
                        presets={presets} savePreset={savePreset} applyPreset={applyPreset} deletePreset={deletePreset}
                    />
                );
            }

            return (
                <UserDashboard 
                    currentUser={currentUser} users={users} stocks={stocks} newsFeed={newsFeed} isTradingLocked={isTradingLocked}
                    onLogout={handleLogout} executeTrade={executeTrade} forceSignoutReason={forceSignoutReason}
                />
            );
        }

        const INITIAL_STOCKS = [
            { symbol: 'NEXA', name: 'Nexa Corp (Tech)', price: 100.00, prevPrice: 100.00 },
            { symbol: 'VRTX', name: 'Vertex Systems (Cyber)', price: 150.00, prevPrice: 150.00 },
            { symbol: 'QDYN', name: 'Quantum Energy', price: 50.00, prevPrice: 50.00 },
            { symbol: 'AERO', name: 'AeroSpace Systems', price: 200.00, prevPrice: 200.00 },
            { symbol: 'ECHO', name: 'Echo Logistics', price: 25.00, prevPrice: 25.00 },
        ];

        const root = ReactDOM.createRoot(document.getElementById('root'));
        root.render(<App />);
    </script>
